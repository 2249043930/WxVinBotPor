# WxVinBot Por 后端开发提示词（Python）

## 一、项目概述

WxVinBot Por 是一个基于微信群的汽车配件询价管理系统后端，采用Python开发。核心功能包括：异步微信消息处理（千寻por框架）、车架号图片识别（LLM多模态+豆包OCR）、VIN查询调度、日志管理、数据管理、RESTful API接口。

**数据库**: 仅使用MySQL 8.0，不使用Redis、Celery等额外组件。

## 二、技术栈要求

| 技术项 | 要求 | 说明 |
|--------|------|------|
| 编程语言 | Python | 3.10+ |
| Web框架 | FastAPI | 异步Web框架 |
| 数据库 | MySQL 8.0 | 唯一数据库 |
| 数据库驱动 | PyMySQL / aiomysql | 异步连接 |
| ORM | SQLAlchemy 2.0 | 异步ORM |
| 迁移工具 | Alembic | 数据库迁移 |
| 微信框架 | 千寻por框架 | 异步微信机器人 |

| OCR识别 | 豆包模型 API | AI OCR识别（LLM fallback），无需传统OCR库 |
| 图片处理 | Pillow | 图片预处理 |
| 日志 | 自定义异步日志 | 按天存储 |
| 配置 | pydantic-settings | 配置管理 |

### 5.1 千寻por微信机器人模块 (modules/wxbot/)

#### 5.1.1 机器人主类 (bot.py)

```python
import asyncio
from typing import Callable, Dict, Any, List
from loguru import logger

class QianxunBot:
    """千寻por微信机器人主类"""
    
    def __init__(self):
        self.bot = None
        self.message_handlers: Dict[str, Callable] = {}
        self.is_running = False
        self.group_members: Dict[str, List[dict]] = {}  # 群成员缓存
        self.wx_id = "wxid_nwrplpmxrwyq22"  # 微信ID
        self.api_base = "http://127.0.0.1:7777"  # 千寻por API地址
        
    async def initialize(self):
        """初始化千寻por机器人"""
        logger.info("千寻por机器人初始化完成")
        
    async def start(self):
        """启动机器人"""
        self.is_running = True
        logger.info("千寻por微信机器人启动")
        # 启动消息监听循环
        await self._message_loop()
        
    async def stop(self):
        """停止机器人"""
        self.is_running = False
        logger.info("千寻por微信机器人停止")
        
    async def _message_loop(self):
        """消息监听循环 - 异步多进程群聊信息监听"""
        while self.is_running:
            try:
                # 获取新消息
                messages = await self._fetch_messages()
                for msg in messages:
                    await self._handle_message(msg)
            except Exception as e:
                logger.error(f"消息处理错误: {e}")
            await asyncio.sleep(0.1)
            
    async def _fetch_messages(self) -> list:
        """获取新消息"""
        # 调用千寻por API获取消息
        return []
        
    async def _handle_message(self, msg: Dict[str, Any]):
        """处理单条消息"""
        msg_type = msg.get("type")
        handler = self.message_handlers.get(msg_type)
        if handler:
            await handler(msg)
            
    def register_handler(self, msg_type: str, handler: Callable):
        """注册消息处理器"""
        self.message_handlers[msg_type] = handler
        
    async def send_text_message(self, to_wxid: str, content: str):
        """发送文本消息"""
        import httpx
        import json
        
        payload = {
            "type": "sendText",
            "data": {
                "wxid": to_wxid,
                "msg": content
            }
        }
        
        async with httpx.AsyncClient() as client:
            response = await client.post(
                f"{self.api_base}/qianxun/httpapi?wxid={self.wx_id}&safekey=",
                json=payload,
                headers={"Content-Type": "application/json"}
            )
            return response.json()
        
    async def send_at_message(self, group_id: str, at_wxid: str, content: str, at_nick: str = "对象昵称"):
        """发送@消息"""
        import httpx
        import json
        
        at_content = f"[@,wxid={at_wxid},nick={at_nick},isAuto=true]\n{content}"
        
        payload = {
            "type": "sendText",
            "data": {
                "wxid": group_id,
                "msg": at_content
            }
        }
        
        async with httpx.AsyncClient() as client:
            response = await client.post(
                f"{self.api_base}/qianxun/httpapi?wxid={self.wx_id}&safekey=",
                json=payload,
                headers={"Content-Type": "application/json"}
            )
            return response.json()
        
    async def get_group_list(self) -> List[dict]:
        """获取微信群聊列表"""
        import httpx
        
        payload = {
            "type": "getGroupList",
            "data": {"type": "1"}
        }
        
        async with httpx.AsyncClient() as client:
            response = await client.post(
                f"{self.api_base}/qianxun/httpapi?wxid={self.wx_id}&safekey=",
                json=payload,
                headers={"Content-Type": "application/json"}
            )
            data = response.json()
            return data.get("result", [])
        
    async def get_group_members(self, group_id: str) -> List[dict]:
        """获取群成员列表"""
        import httpx
        
        payload = {
            "type": "getMemberList",
            "data": {
                "wxid": group_id,
                "type": "1",
                "getNick": "1"
            }
        }
        
        async with httpx.AsyncClient() as client:
            response = await client.post(
                f"{self.api_base}/qianxun/httpapi?wxid={self.wx_id}&safekey=",
                json=payload,
                headers={"Content-Type": "application/json"}
            )
            data = response.json()
            members = data.get("result", [])
            self.group_members[group_id] = members
            return members
        
    async def get_member_nick(self, group_id: str, member_wxid: str) -> str:
        """获取群成员昵称"""
        import httpx
        
        payload = {
            "type": "getMemberNick",
```

```python
class VinRecognizer:
    """车架号识别器 - LLM多模态 + 豆包OCR"""
    
    def __init__(self):
        self.llm_client = LLMClient()
        self.ocr_client = OCRClient()
        self.image_processor = ImageProcessor()
        
    async def recognize(self, image_data: bytes) -> Dict[str, Any]:
        """
        识别图片中的车架号
        
        识别流程：
        1. 优先使用LLM多模态识别
        2. LLM失败时 fallback 到豆包OCR
        
        Args:
            image_data: 图片二进制数据
            
        Returns:
            {
                "vin_code": "识别的VIN码",
                "car_model": "车型信息",
                "confidence": 置信度,
                "success": 是否成功,
                "method": "识别方法：llm/doubao"
            }
        """
        try:
            # 1. 图片预处理
            processed_image = await self.image_processor.process(image_data)
            
            # 2. 优先使用LLM多模态识别
            llm_result = await self.llm_client.recognize_vin(processed_image)
            
            # 3. 检测返回值是否准确
            if self._validate_vin_result(llm_result):
                llm_result["success"] = True
                llm_result["method"] = "llm"
                await self._update_stats(success=True)
                return llm_result
            
            # 4. LLM失败，使用豆包OCR
            logger.warning("LLM识别失败，尝试豆包OCR识别")
            ocr_result = await self.ocr_client.recognize(processed_image)
            
            if self._validate_vin_result(ocr_result):
                ocr_result["success"] = True
                ocr_result["method"] = "doubao"
                await self._update_stats(success=True)
                return ocr_result
            
            # 5. 都失败
            await self._update_stats(success=False)
            return {
                "success": False,
                "method": "none",
                "error": "无法识别有效的VIN码"
            }
            
        except Exception as e:
            logger.error(f"VIN识别失败: {e}")
            await self._update_stats(success=False)
            return {"success": False, "error": str(e)}
            
    def _validate_vin_result(self, result: Dict[str, Any]) -> bool:
        """检测返回值是否准确"""
        vin_code = result.get("vin_code")
        if not vin_code or len(vin_code) != 17:
            return False
        return self._validate_vin_checksum(vin_code)
        
    def _validate_vin_checksum(self, vin_code: str) -> bool:
        """验证VIN码校验位"""
        if len(vin_code) != 17:
            return False
            
        # VIN校验位验证（第9位）
        weights = [8, 7, 6, 5, 4, 3, 2, 10, 0, 9, 8, 7, 6, 5, 4, 3, 2]
        transliterations = {
            'A': 1, 'B': 2, 'C': 3, 'D': 4, 'E': 5, 'F': 6, 'G': 7, 'H': 8,
            'J': 1, 'K': 2, 'L': 3, 'M': 4, 'N': 5, 'P': 7, 'R': 9,
            'S': 2, 'T': 3, 'U': 4, 'V': 5, 'W': 6, 'X': 7, 'Y': 8, 'Z': 9
        }
        
        try:
            total = 0
            for i, char in enumerate(vin_code.upper()):
                if char.isdigit():
                    value = int(char)
                elif char in transliterations:
                    value = transliterations[char]
                else:
                    return False
                total += value * weights[i]
                
            check_digit = total % 11
            if check_digit == 10:
                check_digit = 'X'
            else:
                check_digit = str(check_digit)
                
            return vin_code[8].upper() == str(check_digit).upper()
            
        except Exception:
            return False
```

```python
                raw_result = data["choices"][0]["message"]["content"].strip()

        import re
        
        if not text:
            return None
            
        text = text.upper()
        
        # 检查失败标记
        if any(mark in text for mark in ['FAIL', 'ERROR', '失败', '错误', '无法', 'NULL', 'RECOGNITION_FAILED']):
            return None
            
        # 去除所有非字母数字字符
        text = re.sub(r'[^A-Z0-9]', '', text)
        
        # 提取有效VIN字符（排除I,O,Q）
        valid_chars = re.findall(r'[A-HJ-NPR-Z0-9]', text)
        vin_candidate = ''.join(valid_chars)
        
        # 严格17位
        if len(vin_candidate) == 17:
            return vin_candidate
            
        # 尝试从原文中提取17位连续字符
        matches = re.findall(r'[A-HJ-NPR-Z0-9]{17}', text)
        if matches:
            return matches[0]
            
        return None
```

### 5.4 异步日志模块 (modules/logger/)

#### 5.4.1 异步日志管理器 (async_logger.py)

```python
import asyncio
from datetime import datetime
from enum import Enum
from typing import Optional, List
from loguru import logger
from app.db.session import async_session
from app.models.operation_log import OperationLog

class LogLevel(str, Enum):
    """日志级别"""
    INFO = "INFO"
    WARNING = "WARNING"
    ERROR = "ERROR"
    DEBUG = "DEBUG"

class AsyncLogger:
    """异步日志管理器 - 支持INFO/WARNING/ERROR/LOGS分级存储"""
    
    def __init__(self):
        self.log_queue = asyncio.Queue()
        self.is_running = False
        self._setup_file_logger()
        
    def _setup_file_logger(self):
        """配置文件日志 - 按天为单位存储"""
        logger.add(
            "logs/app_{time:YYYY-MM-DD}.log",
            rotation="00:00",  # 每天零点切割
            retention="30 days",  # 保留30天
            level="INFO",
            encoding="utf-8",
            format="{time:YYYY-MM-DD HH:mm:ss} | {level} | {message}"
        )
        
        # 按级别分文件存储
        logger.add(
            "logs/error_{time:YYYY-MM-DD}.log",
            rotation="00:00",
            retention="30 days",
            level="ERROR",
            encoding="utf-8",
            format="{time:YYYY-MM-DD HH:mm:ss} | {level} | {message}"
        )
        
    async def start(self):
        """启动日志处理器"""
        self.is_running = True
        asyncio.create_task(self._log_processor())
        
    async def stop(self):
        """停止日志处理器"""
        self.is_running = False
        
    async def _log_processor(self):
        """日志处理循环 - 异步数据库日志提交"""
        while self.is_running:
            try:
                # 批量获取日志
                logs = []
                try:
                    while len(logs) < 10:
                        log = await asyncio.wait_for(self.log_queue.get(), timeout=0.1)
                        logs.append(log)
                except asyncio.TimeoutError:
                    pass
                
                # 批量写入数据库
                if logs:
                    await self._batch_save_to_db(logs)
                    
            except Exception as e:
                logger.error(f"日志处理错误: {e}")
                
            await asyncio.sleep(0.1)
            
    async def log(
        self,
        level: LogLevel,
        message: str,
        user_id: Optional[int] = None,
        nickname: Optional[str] = None,
        ip_address: Optional[str] = None,
        login_location: Optional[str] = None,
        os: Optional[str] = None,
        browser: Optional[str] = None,
        operation: Optional[str] = None,
        operation_type: Optional[str] = None,
        extra: Optional[dict] = None
    ):
        """记录日志"""
        log_data = {
            "level": level.value,
            "message": message,
            "nickname": nickname,
            "ip_address": ip_address,
            "login_location": login_location,
            "os": os,
            "browser": browser,
            "login_time": datetime.utcnow(),
            "operation": operation,
            "operation_type": operation_type,
            "extra": extra or {}
        }
        
        # 写入文件日志
        if level == LogLevel.ERROR:
            logger.error(message)
        elif level == LogLevel.WARNING:
            logger.warning(message)
        else:
            logger.info(message)
            
        # 加入队列，异步写入数据库
        await self.log_queue.put(log_data)
        
    async def _batch_save_to_db(self, logs: List[dict]):
        """批量保存日志到数据库"""
        async with async_session() as session:
            try:
                for log_data in logs:
                    log_entry = OperationLog(**{
                        k: v for k, v in log_data.items() 
                        if k != "level" and k != "extra"
                    })
                    session.add(log_entry)
                await session.commit()
            except Exception as e:
                await session.rollback()
                logger.error(f"日志保存到数据库失败: {e}")
                
    # 便捷方法
    async def info(self, message: str, **kwargs):
        """INFO级别日志"""
        await self.log(LogLevel.INFO, message, **kwargs)
        
    async def warning(self, message: str, **kwargs):
        """WARNING级别日志"""
        await self.log(LogLevel.WARNING, message, **kwargs)
        
    async def error(self, message: str, **kwargs):
        """ERROR级别日志"""
        await self.log(LogLevel.ERROR, message, **kwargs)
```

```python
from sqlalchemy.ext.asyncio import create_async_engine, AsyncSession, async_sessionmaker

    autocommit=False,
    autoflush=False
)

async def get_db() -> AsyncSession:
    """获取数据库会话（依赖注入用）"""

            await session.rollback()
            raise
        finally:
            await session.close()
```

## 十、安全模块 (core/security.py)

```python
from datetime import datetime, timedelta
from typing import Optional
from jose import JWTError, jwt
from passlib.context import CryptContext
from fastapi import Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from sqlalchemy.ext.asyncio import AsyncSession
from app.config import settings
from app.db.session import get_db

# 密码加密上下文
pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

# OAuth2 scheme
oauth2_scheme = OAuth2PasswordBearer(tokenUrl="/api/v1/auth/login")

def verify_password(plain_password: str, hashed_password: str) -> bool:
    """验证密码"""
    return pwd_context.verify(plain_password, hashed_password)

def get_password_hash(password: str) -> str:
    """获取密码哈希"""
    return pwd_context.hash(password)

def create_access_token(user_id: int, expires_delta: Optional[timedelta] = None) -> str:
    """创建访问令牌 - 会话保持"""
    if expires_delta:
        expire = datetime.utcnow() + expires_delta
    else:
        expire = datetime.utcnow() + timedelta(minutes=settings.JWT_ACCESS_TOKEN_EXPIRE_MINUTES)
    
    to_encode = {"sub": str(user_id), "exp": expire}
    encoded_jwt = jwt.encode(to_encode, settings.JWT_SECRET_KEY, algorithm=settings.JWT_ALGORITHM)
    return encoded_jwt

async def get_current_user(
    token: str = Depends(oauth2_scheme),
    db: AsyncSession = Depends(get_db)
):
    """获取当前用户 - 账户验证"""
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="无效的认证凭据",
        headers={"WWW-Authenticate": "Bearer"},
    )
    
    try:
        payload = jwt.decode(token, settings.JWT_SECRET_KEY, algorithms=[settings.JWT_ALGORITHM])
        user_id: str = payload.get("sub")
        if user_id is None:
            raise credentials_exception
    except JWTError:
        raise credentials_exception
    
    # 查询用户
    from app.crud.user import user_crud
    user = await user_crud.get(db, id=int(user_id))
    if user is None:
        raise credentials_exception
    
    return user
```

## 十三、开发注意事项

1. **仅使用MySQL**：不使用Redis、Celery等额外组件，所有数据存储和缓存都通过MySQL实现
2. **异步编程**：所有IO操作必须使用async/await，避免阻塞事件循环
3. **千寻por框架**：微信机器人基于千寻por框架实现，需熟悉其API接口
4. **VIN识别**：优先使用LLM多模态识别，失败时fallback到豆包OCR模型
5. **VIN查询**：优先从本地数据库查询，查询不到时调用极速数据API
6. **日志管理**：支持INFO/WARNING/ERROR分级，按天存储到文件，异步写入数据库
7. **数据库连接**：使用连接池，确保会话正确关闭
8. **错误处理**：统一异常处理，返回友好的错误信息
9. **安全配置**：敏感信息（API Key、密码）不得硬编码，使用环境变量
10. **数据验证**：所有输入数据必须经过Pydantic验证
11. **权限控制**：API接口必须验证用户身份和权限
12. **代码规范**：遵循PEP8规范，使用类型注解
13. **VIN校验**：识别结果必须通过校验位验证，确保准确性
14. **消息格式**：发送的微信消息格式需与原项目保持一致
